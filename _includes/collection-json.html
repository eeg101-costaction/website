{%- comment -%} A JSON array of a collection's items for the page scripts: "id" (the item's filename, which is its stable id) plus the named front-matter fields, null when absent. Times go through clock.html (see there). Every "<" is written as \u003c, which reads back as the same text but means editor text containing "" cannot end the script block the JSON sits in. Usage: {% include collection-json.html items=site.events fields="title,start_date" %} {%- endcomment -%} {%- assign __fields = include.fields | split: "," -%} [{%- for __item in include.items -%}{"id":{{ __item.slug | jsonify | replace: "<", "\u003c" }}{%- for __f in __fields -%}{%- assign __key = __f | strip -%}{%- assign __v = __item[__key] -%}{%- if __v != nil and __key == "time" or __v != nil and __key == "end_time" -%}{%- capture __c -%}{%- include clock.html value=__v -%}{%- endcapture -%}{%- assign __v = __c -%}{%- endif -%},{{ __key | jsonify }}:{{ __v | jsonify | replace: "<", "\u003c" }}{%- endfor -%}}{%- unless forloop.last -%},{%- endunless -%}{%- endfor -%}]
